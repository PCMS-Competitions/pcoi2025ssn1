    \section*{\fontsize{24}{28}\selectfont\taskname}
    設 $a,b,A,B,n$ 有以下關係式\\
    $\quad \cdot \quad A=na-b$\\
    $\quad \cdot \quad B=n^2a-(n-1)b$\\
    給定 $A,B,n$ ，求 $a$ 和 $b$
    \section*{實現細節}
        你需要在標準輸入（\texttt{stdin}）讀入樣例，並在標準輸出（\texttt{stdout}）輸出答案。
    \subsection*{輸入格式}
    \begin{codeblock}
        $A\quad B \quad n$
    \end{codeblock}
    \subsection*{輸出格式}
    \begin{codeblock}
        $a\quad b$
    \end{codeblock}
    
    \section*{例子}
    考慮以下樣例： 
    \begin{codeblock}
        $5 \quad 11 \quad 2$
    \end{codeblock}
    你的程序應輸出：
    \begin{codeblock}
        $3\quad 1$
    \end{codeblock}
    \section*{約束條件}
    \begin{itemize}
        \item $-1e18 \le a,b,A,B,n \le 1e18$
    \end{itemize}
    \section*{子任務}
    \begin{enumerate}
        \item （40分）保證 $a,b,n$ 的絕對值不大於 $1000$
        \item （60分）没有額外的約束條件
    \end{enumerate}
