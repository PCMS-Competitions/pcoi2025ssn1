    \section*{\fontsize{24}{28}\selectfont\taskname}
    給定 $m$ 個區間 $[l_i,\ r_i]$ 和 $n$ 個顔色 ${c_j}$\\
    求下標 $j$ 為 $[l_i,\ r_i]$ 區間内的 $c_j$ 按序組成了多少個顔色段 $cnt_i$
    \section*{實現細節}
        你需要在標準輸入（\texttt{stdin}）讀入樣例，並在標準輸出（\texttt{stdout}）輸出答案。
    \subsection*{輸入格式}
    \begin{codeblock}
        $n\quad m$\\
        $l_1 \quad r_1$\\
        $l_2 \quad r_2$\\
        $\vdots $\\
        $l_m \quad r_m$\\
        $c_1 \quad c_2 \quad \dots \quad c_n$
    \end{codeblock}
    \subsection*{輸出格式}
    \begin{codeblock}
        $cnt_1$\\
        $cnt_2$\\
        $\vdots$\\
        $cnt_m$
    \end{codeblock}
    
    \section*{例子}
    考慮以下樣例： 
    \begin{codeblock}
        $5\quad 2$\\
        $1 \quad 4$\\
        $2 \quad 5$\\
        $1 \quad 1\quad 2\quad 2\quad 1$
    \end{codeblock}
    你的程序應輸出：
    \begin{codeblock}
        $2$\\
        $3$
    \end{codeblock}
    \section*{約束條件}
    \begin{itemize}
        \item $1 \le n,m \le 200000$
        \item $1\le c_i \le 1e9$
        \item $1 \le l_i \le r_i\le n$
    \end{itemize}
    \section*{子任務}
    \begin{enumerate}
        \item （40分）$1 \le n,m \le 1000$
        \item （60分）没有額外的約束條件
    \end{enumerate}
